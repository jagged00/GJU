sect_spar_interaction = {
    category = interaction_category_friendly
    interface_priority = 120
    common_interaction = yes
    icon = icon_combat

    popup_on_receive = yes
    pause_on_receive = yes

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:actor = scope:recipient
                }
                desc = sect_spar_interaction_desc_own_court
            }
            desc = sect_spar_interaction_desc_other_court
        }
    }

    is_valid_showing_failures_only = {
        scope:secondary_actor = {
            is_imprisoned = no
            is_incapable = no
            is_adult = yes
            is_alive = yes
        }
        scope:secondary_recipient = {
            is_imprisoned = no
            is_incapable = no
            is_adult = yes
            is_alive = yes
        }
        custom_description = {
            text = cannot_spar_yourself
            NOT = { scope:secondary_actor = scope:secondary_recipient }
        }
    }

    populate_actor_list = {
        scope:actor = {
            add_to_list = characters

            every_courtier = {
                limit = {
                    is_adult = yes
                    is_alive = yes
                    is_imprisoned = no
                    is_incapable = no
                    NOT = { this = scope:actor }
                }
                add_to_list = characters
            }

            every_child = {
                limit = {
                    is_adult = yes
                    is_alive = yes
                    is_imprisoned = no
                    is_incapable = no
                    NOT = { is_in_list = characters }
                }
                add_to_list = characters
            }
        }
    }

    populate_recipient_list = {
        scope:recipient = {
            add_to_list = characters

            every_courtier = {
                limit = {
                    is_adult = yes
                    is_alive = yes
                    is_imprisoned = no
                    is_incapable = no
                    NOT = { this = scope:recipient }
                }
                add_to_list = characters
            }

            every_child = {
                limit = {
                    is_adult = yes
                    is_alive = yes
                    is_imprisoned = no
                    is_incapable = no
                    NOT = { is_in_list = characters }
                }
                add_to_list = characters
            }
        }
    }

    on_accept = {
        scope:secondary_actor = {
            if = {
                limit = { NOT = { exists = var:sect_spars_won } }
                set_variable = { name = sect_spars_won value = 0 }
            }
            if = {
                limit = { NOT = { exists = var:sect_spars_lost } }
                set_variable = { name = sect_spars_lost value = 0 }
            }
            if = {
                limit = { NOT = { exists = var:sect_spars_total } }
                set_variable = { name = sect_spars_total value = 0 }
            }
            custom_tooltip = cultivation_spar_effect.tt

            scope:secondary_recipient = {
                save_scope_as = sparring_partner

                if = {
                    limit = { NOT = { exists = var:sect_spars_won } }
                    set_variable = { name = sect_spars_won value = 0 }
                }
                if = {
                    limit = { NOT = { exists = var:sect_spars_lost } }
                    set_variable = { name = sect_spars_lost value = 0 }
                }
                if = {
                    limit = { NOT = { exists = var:sect_spars_total } }
                    set_variable = { name = sect_spars_total value = 0 }
                }
                custom_tooltip = cultivation_spar_effect.ttt
            }

            trigger_event = cultivation_sect_spar.1000
        }
    }

    send_option = {
        is_shown = {
            scope:actor = scope:recipient
        }
        is_valid = {
            scope:actor = {
                OR = {
                    this = scope:secondary_actor
                    has_courtier = scope:secondary_actor
                }
            }
            scope:recipient = {
                OR = {
                    this = scope:secondary_recipient
                    has_courtier = scope:secondary_recipient
                }
            }
        }
        flag = send_own_court
        localization = SECT_SPAR_SEND_OWN_COURT
    }

    send_option = {
        is_shown = {
            NOT = { scope:actor = scope:recipient }
        }
        is_valid = {
            scope:actor = {
                OR = {
                    this = scope:secondary_actor
                    has_courtier = scope:secondary_actor
                }
            }
            scope:recipient = {
                OR = {
                    this = scope:secondary_recipient
                    has_courtier = scope:secondary_recipient
                }
            }
        }
        flag = send_other_court
        localization = SECT_SPAR_SEND_OTHER_COURT
    }

    ai_accept = {
        base = 50

        modifier = {
            add = 25
            scope:secondary_recipient = {
                prowess >= 12
            }
            desc = AI_PROWESS_REASON
        }

        modifier = {
            add = -25
            scope:secondary_recipient = {
                prowess < 8
            }
            desc = AI_PROWESS_REASON
        }
    }

    ai_will_do = {
        base = 0
    }
}
